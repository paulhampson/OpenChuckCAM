For points inside the cutter, signed clearance is max(r - 1, -z).

#![allow(dead_code, clippy::print_stdout, clippy::cast_precision_loss, clippy::cast_possible_truncation, clippy::cast_sign_loss)]

mod measure;
use openchuck_types::brep::{BrepFace, BrepModel, FaceId, Frame, Surface, SurfaceKind};
use openchuck_types::geometry::{Point3, Vec3};
use openchuck_types::mesh::TriangleMesh;
use openchuck_types::tool::{Tool, ToolShape};
use openchuck_types::units::Millimetres;
fn main() -> Result<(), Box<dyn std::error::Error>> {
    let s=0.5_f64.sqrt();
    let face=BrepFace {
        id: FaceId(0), surface: SurfaceKind::Plane, body: 0, edges: vec![], reversed: false,
        geometry: Surface::Plane { frame: Frame { origin: Point3::new(0.95,0.0,-0.03),
            x:Vec3::new(s,0.0,s), y:Vec3::new(0.0,1.0,0.0), z:Vec3::new(-s,0.0,s) } },
        uv_bounds:[0.0, 0.1*2.0_f64.sqrt(),0.0,0.05], trims:vec![],
        tessellation:TriangleMesh { vertices:vec![Point3::new(0.95,0.0,-0.03),Point3::new(1.05,0.0,0.07),Point3::new(1.05,0.05,0.07),Point3::new(0.95,0.05,-0.03)], triangles:vec![[0,1,2],[0,2,3]] }
    };
    let model=BrepModel { faces:vec![face], ..BrepModel::default() };
    let tool=Tool { shape:ToolShape::FlatEnd, diameter:Millimetres(2.0), flute_length:Millimetres(8.0), overall_length:Millimetres(40.0), shank_diameter:Millimetres(2.0), flute_count:2, material:None, centre_cutting:Some(true) };
    let outline=openchuck_testkit::gouge::Outline::new(&tool.profile()?);
    // On z=x-0.98, y>=0, r>=x. The signed clearance inside a unit-radius
    // flat cutter is max(r-1,-z) >= max(x-1,0.98-x), minimized at x=0.99.
    // (0.99,0,0.01) is on this face and attains exactly -0.01 mm.
    for spacing in [0.05,0.025,0.02,0.01,0.005,0.0005] {
        let cloud=measure::Cloud::of_faces(&model,spacing);
        let measured=cloud.clearance(&outline,Point3::ORIGIN,Vec3::new(0.0,0.0,1.0),8.0,1.0);
        println!("spacing {spacing:.6} | points {} | sampled clearance {measured:.9} | analytic clearance -0.010000000 | missed depth {:.9}",cloud.len(),measured+0.01);
    }
    Ok(())
}
