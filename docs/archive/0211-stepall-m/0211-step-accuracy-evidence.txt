use openchuck_app::{Application, PostSelection, Threads};
use openchuck_types::kinematics::{MachinePosition, RotaryFrame, machine_xyz, tool_tip_in_model};
use openchuck_types::geometry::Point3;
use openchuck_types::units::Degrees;

fn main() -> Result<(), Box<dyn std::error::Error>> {
    for n in [4, 2] {
        let mut app = Application::open(std::path::Path::new("/workspace/stepall-evidence/project.stl"), Threads::Exactly(n))?;
        app.set_post(Some(PostSelection::parse("makera-z1")));
        let result = app.regenerate()?;
        let setup = app.project().setups().first().ok_or("no setup")?;
        let frame = RotaryFrame::work(app.project().machine().ok_or("no machine")?, &setup.frame);
        let (_, text) = result.gcode.first().ok_or("no G-code")?;
        std::fs::write(format!("/workspace/stepall-evidence/form-a-probe-{n}.nc"), text)?;
        for report in &result.reports {
            println!("{n} threads: findings {:?}; comparison {:?}", report.findings, report.comparison);
        }
        let mut worst = 0.0_f64;
        let mut detail = serde_json::Value::Null;
        let replay = openchuck_gcode::replay(&openchuck_gcode::read(text).map_err(|e| format!("{e:?}"))?, openchuck_gcode::Dialect::MakeraZ1)?;
        for (g, group) in result.toolpath.groups.iter().enumerate() {
            for (i, pose) in std::iter::once(&group.start).chain(group.segments.iter().map(|s| &s.end)).enumerate() {
                let xyz = machine_xyz(&frame, pose.tip, pose.a);
                let rounded = Point3::new(openchuck_post::round_written(xyz.x, 3), openchuck_post::round_written(xyz.y, 3), openchuck_post::round_written(xyz.z, 3));
                let a = openchuck_post::round_written(pose.a.0, 3);
                let back = tool_tip_in_model(&frame, &MachinePosition {xyz: rounded, a: Degrees(a)});
                let error = back.distance(pose.tip);
                if error > worst {
                    worst = error;
                    let lines: Vec<_> = replay.moves().filter(|m| !m.machine && (m.to[0]-rounded.x).abs()<1e-8 && (m.to[1]-rounded.y).abs()<1e-8 && (m.to[2]-rounded.z).abs()<1e-8 && ((m.to[3]-a)/360.0 - ((m.to[3]-a)/360.0).round()).abs()<1e-8).map(|m| m.line).collect();
                    detail = serde_json::json!({"group_index":g,"operation_tag":group.tags.operation,"pose_index":i,"tip":pose.tip,"a_exact":pose.a.0,"xyz_exact":xyz,"xyz_rounded":rounded,"a_rounded":a,"xyz_rounding_delta":[rounded.x-xyz.x,rounded.y-xyz.y,rounded.z-xyz.z],"a_rounding_delta_deg":a-pose.a.0,"rounding_only_tip_error_mm":error,"gcode_lines":lines});
                }
            }
        }
        println!("{n} threads rounding-only worst:\n{}", serde_json::to_string_pretty(&detail)?);
        std::fs::write(format!("/workspace/stepall-evidence/form-a-rounding-{n}.json"), serde_json::to_vec_pretty(&detail)?)?;
    }
    Ok(())
}
