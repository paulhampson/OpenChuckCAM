== 4. Five-run qualification of 0.0025 mm / 0.015 rad (the approved method)
Runner script (each thread count's b/c pair back to back on one host; a pair interrupted by a restart is rerun whole):
#!/bin/bash
# Each thread count's b/c pair runs back to back on one host; a pair left
# unfinished by a restart is rerun whole (its partial logs are archived).
cd /home/user/OpenChuckCAM
export OPENCHUCK_STEP_WORKER=$PWD/target/release/openchuck-step-worker RAYON_NUM_THREADS=4
host="$(grep -m1 'model name' /proc/cpuinfo | cut -d: -f2) boot $(cat /proc/sys/kernel/random/boot_id)"
for threads in all 1; do
  [ -f <scratch>/m3/pair-$threads.done ] && continue
  rm -f <scratch>/m3/m3-?-$threads.log*
  for form in b c; do
    log=<scratch>/m3/m3-$form-$threads.log
    echo "start $form $threads $(date -u) host$host" >> <scratch>/m3/status.txt
    <scratch>/m3/m3-step-accuracy --only m3 --part blisk --forms $form --ops roughing,rest,pencil,helical,axial,indexed --runs 5 --threads $threads --skip-probe --fine 0.0025 --angular 0.015 > $log 2>&1
    code=$?; echo "end $form $threads exit=$code $(date -u)" >> <scratch>/m3/status.txt
    [ $code -eq 0 ] || exit 1
  done
  echo "$host" > <scratch>/m3/pair-$threads.done
done
echo COMPLETE >> <scratch>/m3/status.txt
